CLASS ltcl_unit_test IMPLEMENTATION.
  METHOD setup.
    test_instance = NEW #( ).
    user = NEW #( ).
    order = NEW #( amount = 2000 ).
    order->items = VALUE #( ( NEW #( name  = |Keyboard|
                                     price = 100 ) )
                            ( NEW #( name  = |Mouse|
                                     price = 50 ) ) ).
  ENDMETHOD.

  METHOD shall_no_order_be_rejected.
    " test_invalid_order_raises_exception_is_caught
    " given
    DATA(approval_status_exp) = |rejected|.

    FREE order.
    check_approval_status( approval_status_exp ).
  ENDMETHOD.

  METHOD shall_no_user_be_rejected.
    " test_invalid_order_raises_exception_is_caught
    " given
    DATA(approval_status_exp) = |rejected|.

    FREE user.
    check_approval_status( approval_status_exp ).
  ENDMETHOD.

  METHOD teardown.
    FREE: user,
          order,
          test_instance.
  ENDMETHOD.

  METHOD check_approval_status.
    " when
    DATA(approval_status_act) = test_instance->approve_order( order = order
                                                              user  = user ).

    " then
    cl_abap_unit_assert=>assert_equals( exp = approval_status_exp
                                        act = approval_status_act ).
  ENDMETHOD.

  METHOD shall_all_combinations_be_ok.
    TYPES: BEGIN OF ts_combination,
             negativ_item_price  TYPE abap_bool,
             is_premium          TYPE abap_bool,
             amount_greater_1000 TYPE abap_bool,
             has_discount        TYPE abap_bool,
             user_region         TYPE string,
             currency            TYPE string,
             order_type          TYPE string,
             is_trial            TYPE abap_bool,
             is_admin            TYPE abap_bool,
             approval_status     TYPE string,
           END OF ts_combination.
    TYPES tt_combinations TYPE STANDARD TABLE OF ts_combination WITH EMPTY KEY.
    TYPES tt_booleans     TYPE STANDARD TABLE OF abap_bool WITH EMPTY KEY.

    " given
    DATA(negativ_order_item_added) = abap_false.
    DATA(combinations) = VALUE tt_combinations( LET booleans     = VALUE tt_booleans( ( abap_false )
                                                                                      ( abap_true ) )
                                                    user_regions = VALUE string_table( ( |EU| )
                                                                                       ( |US| ) )
                                                    currencies   = VALUE string_table( ( |EUR| )
                                                                                       ( |USD| ) )
                                                    order_types  = VALUE string_table( ( |bulk| )
                                                                                       ( |normal| ) ) IN
                                                FOR <negativ_item_price> IN booleans
                                                FOR <is_premium> IN booleans
                                                FOR <amount_greater_1000> IN booleans
                                                FOR <has_discount> IN booleans
                                                FOR <user_region> IN user_regions
                                                FOR <currency> IN currencies
                                                FOR <order_type> IN order_types
                                                FOR <is_trial> IN booleans
                                                FOR <is_admin> IN booleans
                                                ( negativ_item_price  = <negativ_item_price>
                                                  is_premium          = <is_premium>
                                                  amount_greater_1000 = <amount_greater_1000>
                                                  has_discount        = <has_discount>
                                                  user_region         = <user_region>
                                                  currency            = <currency>
                                                  order_type          = <order_type>
                                                  is_trial            = <is_trial>
                                                  is_admin            = <is_admin> ) ).

    LOOP AT combinations ASSIGNING FIELD-SYMBOL(<combination>).

      user->is_admin   = <combination>-is_admin.
      user->is_premium = <combination>-is_premium.
      user->is_trial   = <combination>-is_trial.
      user->region     = <combination>-user_region.
      order->has_discount = <combination>-has_discount.
      order->currency     = <combination>-currency.
      order->type         = <combination>-order_type.
      order->amount       = COND #( WHEN <combination>-amount_greater_1000 = abap_true
                                    THEN 1001
                                    ELSE 1000 ).
      change_order_items( EXPORTING add_negativ_item_price   = <combination>-negativ_item_price
                          CHANGING  negativ_order_item_added = negativ_order_item_added ).
      <combination>-approval_status = test_instance->approve_order_original( order = order
                                                                             user  = user ).

    ENDLOOP.
    LOOP AT combinations ASSIGNING <combination>.

      DATA(line_number) = sy-tabix.

      user->is_admin   = <combination>-is_admin.
      user->is_premium = <combination>-is_premium.
      user->is_trial   = <combination>-is_trial.
      user->region     = <combination>-user_region.
      order->has_discount = <combination>-has_discount.
      order->currency     = <combination>-currency.
      order->type         = <combination>-order_type.
      order->amount       = COND #( WHEN <combination>-amount_greater_1000 = abap_true
                                    THEN 1001
                                    ELSE 1000 ).
      change_order_items( EXPORTING add_negativ_item_price   = <combination>-negativ_item_price
                          CHANGING  negativ_order_item_added = negativ_order_item_added ).

      " when
      DATA(approval_status_act) = test_instance->approve_order( order = order
                                                                user  = user ).

      " then
      cl_abap_unit_assert=>assert_equals( exp  = <combination>-approval_status
                                          act  = approval_status_act
                                          msg  = |Error in line: { line_number }|
                                          quit = if_abap_unit_constant=>quit-no ).

    ENDLOOP.
  ENDMETHOD.

  METHOD change_order_items.
    " has to be deleted
    IF     add_negativ_item_price   = abap_false
       AND negativ_order_item_added = abap_true.

      DELETE order->items INDEX 1.

    ENDIF.
    " has to be added
    IF     add_negativ_item_price   = abap_true
       AND negativ_order_item_added = abap_false.

      INSERT NEW zcldlt_arjan_codes_item( name  = 'negativ_item'
                                          price = -1 )
             INTO order->items INDEX 1.

    ENDIF.
    " set the current state
    negativ_order_item_added = add_negativ_item_price.
  ENDMETHOD.
ENDCLASS.
