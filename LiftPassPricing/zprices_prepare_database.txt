CLASS zprices_prepare_database DEFINITION
  PUBLIC FINAL
  CREATE PUBLIC.

  PUBLIC SECTION.
    INTERFACES if_oo_adt_classrun.

  PRIVATE SECTION.
    METHODS delete_all_prices.
    METHODS delete_all_holidays.
    METHODS set_holidays.
    METHODS set_prices.

    METHODS display_prices
      IMPORTING !out TYPE REF TO if_oo_adt_classrun_out.

    METHODS display_holidays
      IMPORTING !out TYPE REF TO if_oo_adt_classrun_out.

    METHODS add_current_date_to_holidays
      IMPORTING !add TYPE abap_bool.
ENDCLASS.


CLASS zprices_prepare_database IMPLEMENTATION.
  METHOD if_oo_adt_classrun~main.
    DATA(add_current_date) = abap_true.

    delete_all_prices( ).
    set_prices( ).
    delete_all_holidays( ).
    set_holidays( ).
    add_current_date_to_holidays( add = add_current_date ).
    display_prices( out ).
    display_holidays( out ).
  ENDMETHOD.

  METHOD set_prices.
*    INSERT zdltbase_prices FROM TABLE @( VALUE #( ( cost = 35 type = '1jour' )
*                                                  ( cost = 19 type = 'night' ) ) ).
    INSERT zdltbase_prices FROM TABLE @( VALUE #( ( cost = 45 type = '1jour' )
                                                  ( cost = 169 type = 'night' ) ) ).
  ENDMETHOD.

  METHOD set_holidays.
    INSERT zdltholidays FROM TABLE @( VALUE #( description = 'winter'
                                               ( holiday = '20190218' )
                                               ( holiday = '20190225' )
                                               ( holiday = '20190304' ) ) ).
  ENDMETHOD.

  METHOD delete_all_holidays.
    DELETE FROM zdltholidays.
  ENDMETHOD.

  METHOD delete_all_prices.
    DELETE FROM zdltbase_prices.
  ENDMETHOD.

  METHOD display_prices.
    SELECT  type,
            cost
        FROM zdltbase_prices
        INTO TABLE @DATA(prices).
    out->write( prices ).
  ENDMETHOD.

  METHOD display_holidays.
    SELECT holiday,
           description
        FROM zdltholidays
        INTO TABLE @DATA(holidays).
    out->write( holidays ).
  ENDMETHOD.

  METHOD add_current_date_to_holidays.
    CHECK add = abap_true.

    INSERT zdltholidays FROM TABLE @( VALUE #( description = 'for price3'
                                               ( holiday = cl_abap_context_info=>get_system_date( ) ) ) ).
  ENDMETHOD.
ENDCLASS.
